// ==UserScript==
// @name         秒哒每日签到助手
// @namespace    https://www.miaoda.cn/
// @version      1.0.0
// @description  每天首次访问秒哒应用时，自动打开签到页面领取100积分（每日仅触发一次）
// @author       You
// @match        https://*.appmiaoda.com/*
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_openInTab
// @run-at       document-idle
// ==/UserScript==

(function () {
  'use strict';

  /* ================================================================
   *  打开方式枚举（修改 OPEN_MODE 的值即可切换）
   * ================================================================ */
  const OPEN_MODE = Object.freeze({
    NEW_TAB:     'new_tab',      // 新标签页打开
    POPUP:       'popup',        // 小窗（弹窗）打开
    CURRENT_TAB: 'current_tab',  // 当前标签页跳转
  });

  /* ================================================================
   *  ⚙️ 配置区 —— 只需修改这里
   * ================================================================ */
  const CONFIG = {
    // 👇 选择打开方式：OPEN_MODE.NEW_TAB / OPEN_MODE.POPUP / OPEN_MODE.CURRENT_TAB
    openMode: OPEN_MODE.POPUP,

    // 要打开的签到页面地址
    targetUrl: 'https://www.miaoda.cn/projects/app-cur0nttk7f29/page-d551997s87i9',

    // 小窗模式下的窗口尺寸（仅 POPUP 生效）
    popupWidth:  480,
    popupHeight: 640,

    // 页面加载后延迟多少毫秒再打开（避免被浏览器拦截）
    delayMs: 1500,

    // 是否在新标签页中激活（置前），仅 NEW_TAB 生效
    newTabActive: true,
  };

  /* ================================================================
   *  核心逻辑
   * ================================================================ */

  // 获取今天的日期字符串，如 "2026-08-20"
  function getTodayStr() {
    const d = new Date();
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${y}-${m}-${day}`;
  }

  // 判断今天是否已经打开过
  function hasOpenedToday() {
    return GM_getValue('lastSignDate', '') === getTodayStr();
  }

  // 标记今天已打开
  function markOpened() {
    GM_setValue('lastSignDate', getTodayStr());
  }

  // 根据配置选择打开方式
  function openTarget() {
    const url = CONFIG.targetUrl;

    switch (CONFIG.openMode) {

      case OPEN_MODE.NEW_TAB:
        // 使用 GM_openInTab，不受弹窗拦截影响
        GM_openInTab(url, {
          active: CONFIG.newTabActive,  // true = 切换到新标签页
          insert: true,                 // 插入到当前标签页后面
        });
        break;

      case OPEN_MODE.POPUP: {
        // 计算居中位置
        const left = Math.round((screen.width  - CONFIG.popupWidth)  / 2);
        const top  = Math.round((screen.height - CONFIG.popupHeight) / 2);
        const features = [
          `width=${CONFIG.popupWidth}`,
          `height=${CONFIG.popupHeight}`,
          `left=${left}`,
          `top=${top}`,
          'menubar=no',
          'toolbar=no',
          'location=yes',
          'status=no',
          'scrollbars=yes',
          'resizable=yes',
        ].join(',');
        window.open(url, '_miaoda_sign', features);
        break;
      }

      case OPEN_MODE.CURRENT_TAB:
        window.location.href = url;
        break;

      default:
        console.warn('[秒哒签到] 未知的 openMode，回退到新标签页');
        GM_openInTab(url, { active: true });
    }
  }

  /* ================================================================
   *  入口
   * ================================================================ */
  if (!hasOpenedToday()) {
    console.log(`[秒哒签到] 今日尚未签到，${CONFIG.delayMs}ms 后自动打开签到页…`);

    setTimeout(() => {
      openTarget();
      markOpened();
      console.log('[秒哒签到] ✅ 已打开签到页面，今日不再重复触发。');
    }, CONFIG.delayMs);

  } else {
    console.log('[秒哒签到] 今日已签到，跳过。');
  }

})();
